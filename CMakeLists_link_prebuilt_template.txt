cmake_minimum_required(VERSION 3.22.1)

project(webrtc_apm_jni)

set(USE_PREBUILT_WEBRTC_APM ON)

add_library(
        webrtc_apm_jni
        SHARED
        webrtc_apm_jni.cpp
)

find_library(
        log-lib
        log
)

target_link_libraries(
        webrtc_apm_jni
        ${log-lib}
)

if (USE_PREBUILT_WEBRTC_APM)
    set(WEBRTC_APM_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/webrtc_apm)

    target_compile_definitions(
            webrtc_apm_jni
            PRIVATE
            USE_PREBUILT_WEBRTC_APM=1
    )

    target_include_directories(
            webrtc_apm_jni
            PRIVATE
            ${WEBRTC_APM_ROOT}/include
    )

    add_library(
            webrtc_audio_processing
            STATIC
            IMPORTED
    )

    set_target_properties(
            webrtc_audio_processing
            PROPERTIES
            IMPORTED_LOCATION
            ${WEBRTC_APM_ROOT}/lib/${ANDROID_ABI}/libwebrtc_audio_processing.a
    )

    target_link_libraries(
            webrtc_apm_jni
            webrtc_audio_processing
    )
endif()
